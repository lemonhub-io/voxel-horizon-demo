  uv(t: number): [number, number, number, number] {
    const s = 1 / this.size;
    const u = (t % this.size) * s, v = Math.floor(t / this.size) * s;
    const e = 0.001;
    return [u + e, 1 - v - s + e, u + s - e, 1 - v - e];
  }

    const out = U.rgbHex((r / n) * f, (g / n) * f, (b / n) * f);
